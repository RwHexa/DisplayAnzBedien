/* ==========================================================================
 * ESP32 WROOM <-> NEXTION 2.8 ZOLL DISPLAY (BI-DIREKTIONALE STEUERUNG)
 * Hardware-UART2: RX2 = GPIO 16, TX2 = GPIO 17 (9600 Baud)
 * Externe LED an GPIO 32 mit PWM-Helligkeitssteuerung (Slider h0)
 * USB-UART0: Debugging & Delphi Verbindung (115200 Baud)
 * ========================================================================== */

#include <Arduino.h>

// Pins fuer Nextion Display an HardwareSerial 2 des ESP32
#define NEXTION_RX_PIN 16 // Verbinden mit Nextion TX (Blau)
#define NEXTION_TX_PIN 17 // Verbinden mit Nextion RX (Gelb)
#define NEXTION_BAUD   9600

// Externe LED an GPIO 32 mit Vorwiderstand (z. B. 330 Ohm)
#define LED_PIN 32

HardwareSerial NextionSerial(2); // Nutzt Hardware-UART2 des ESP32

// Globale Zustaende
static bool led_state = true;         // LED Ein / Aus
static int  led_brightness = 50;      // 0 bis 100 % (vom Slider h0)

// -----------------------------------------------------------------------------
// 1. PWM-STEUERUNG FUER DIE LED AN GPIO 32
// -----------------------------------------------------------------------------

void update_led_pwm() {
    if (led_state) {
        // Helligkeit 0..100% auf 8-Bit PWM (0..255) umrechnen
        int pwm_val = map(led_brightness, 0, 100, 0, 255);
        analogWrite(LED_PIN, pwm_val);
        Serial.print("--> [LED GPIO 32] STATUS: AN | Helligkeit: ");
        Serial.print(led_brightness);
        Serial.print("% (PWM: ");
        Serial.print(pwm_val);
        Serial.println("/255)");
    } else {
        analogWrite(LED_PIN, 0); // Ausgeschaltet (0 PWM)
        Serial.println("--> [LED GPIO 32] STATUS: AUS (0 PWM)");
    }
}

// -----------------------------------------------------------------------------
// 2. HILFSFUNKTIONEN ZUM SENDEN AN DAS NEXTION DISPLAY
// -----------------------------------------------------------------------------

// Sendet Befehl + 3x 0xFF Abschluss
void nextion_send_cmd(const String& cmd) {
    NextionSerial.print(cmd);
    NextionSerial.write(0xFF);
    NextionSerial.write(0xFF);
    NextionSerial.write(0xFF);
}

// Textfeld aktualisieren (z.B. t0.txt="23.4 C")
void nextion_set_text(const String& component, const String& text) {
    String cmd = component + ".txt=\"" + text + "\"";
    nextion_send_cmd(cmd);
}

// Zahlenwert aktualisieren (z.B. n0.val=1450, h0.val=50, bt0.val=1)
void nextion_set_number(const String& component, int32_t val) {
    String cmd = component + ".val=" + String(val);
    nextion_send_cmd(cmd);
}

// -----------------------------------------------------------------------------
// 3. EMPFANGS-LOGIK: BUTTONS & SLIDER VOM NEXTION DISPLAY VERARBEITEN
// -----------------------------------------------------------------------------

void handle_nextion_data() {
    static uint8_t rx_buffer[32];
    static uint8_t rx_index = 0;
    static uint8_t ff_count = 0;

    while (NextionSerial.available()) {
        uint8_t b = NextionSerial.read();

        // Auf 3x 0xFF Endekennung pruefen
        if (b == 0xFF) {
            ff_count++;
            if (ff_count == 3) {
                // FALL 1: Touch-Event Paket (0x65 [Page] [CompID] [Event])
                if (rx_index >= 4 && rx_buffer[0] == 0x65) {
                    uint8_t pageId    = rx_buffer[1];
                    uint8_t compId    = rx_buffer[2];
                    uint8_t eventType = rx_buffer[3]; // 1 = Press, 0 = Release

                    Serial.print("[NEXTION TOUCH] Page: ");
                    Serial.print(pageId);
                    Serial.print(" | Component-ID: ");
                    Serial.print(compId);
                    Serial.print(" | Event: ");
                    Serial.println(eventType == 1 ? "PRESS" : "RELEASE");

                    // Wenn Slider h0 beruehrt wurde -> aktuellen Wert mit "get h0.val" anfordern
                    nextion_send_cmd("get h0.val");

                    // Button bt0 schaltet die LED EIN / AUS
                    if (eventType == 1) { // Bei Tastendruck
                        led_state = !led_state;
                        update_led_pwm();

                        // Display-Status aktualisieren
                        if (led_state) {
                            nextion_set_text("t1", "LED: AN (" + String(led_brightness) + "%)");
                            nextion_set_number("bt0", 1);
                        } else {
                            nextion_set_text("t1", "LED: AUS");
                            nextion_set_number("bt0", 0);
                        }
                    }
                }
                // FALL 2: Numerische Rueckmeldung vom Display (0x71 [4 Bytes Integer])
                // Wird von Nextion als Antwort auf "get h0.val" gesendet!
                else if (rx_index >= 5 && rx_buffer[0] == 0x71) {
                    uint32_t val = (uint32_t)rx_buffer[1] | 
                                  ((uint32_t)rx_buffer[2] << 8) | 
                                  ((uint32_t)rx_buffer[3] << 16) | 
                                  ((uint32_t)rx_buffer[4] << 24);

                    led_brightness = constrain((int)val, 0, 100);
                    Serial.print("[NEXTION SLIDER h0] Neuer Helligkeitswert: ");
                    Serial.print(led_brightness);
                    Serial.println("%");

                    // PWM Helligkeit auf GPIO 32 anwenden
                    update_led_pwm();

                    // Wenn LED an ist, Textanzeige mit Prozent aktualisieren
                    if (led_state) {
                        nextion_set_text("t1", "LED: AN (" + String(led_brightness) + "%)");
                    }
                }
                // FALL 3: Direkter ASCII-String (z.B. "H:75" aus Touch Move Event)
                else if (rx_index >= 3 && rx_buffer[0] == 'H' && rx_buffer[1] == ':') {
                    rx_buffer[rx_index] = '\0';
                    int val = atoi((char*)&rx_buffer[2]);
                    led_brightness = constrain(val, 0, 100);
                    update_led_pwm();
                    if (led_state) {
                        nextion_set_text("t1", "LED: AN (" + String(led_brightness) + "%)");
                    }
                }

                // Puffer leeren
                rx_index = 0;
                ff_count = 0;
                continue;
            }
        } else {
            ff_count = 0;
            if (rx_index < sizeof(rx_buffer) - 1) {
                rx_buffer[rx_index++] = b;
            }
        }
    }
}

// -----------------------------------------------------------------------------
// 4. SETUP & HAUPTSCHLEIFE
// -----------------------------------------------------------------------------

void setup() {
    // 1. Debug-Monitor via USB (UART0)
    Serial.begin(115200);

    // 2. LED Pin an GPIO 32 initialisieren
    pinMode(LED_PIN, OUTPUT);
    update_led_pwm(); // Startet mit 50% Helligkeit

    // 3. Hardware-UART2 fuer Nextion initialisieren
    NextionSerial.begin(NEXTION_BAUD, SERIAL_8N1, NEXTION_RX_PIN, NEXTION_TX_PIN);

    delay(500); // Warten auf Display-Start

    Serial.println("=================================================");
    Serial.println(" ESP32 WROOM <-> Nextion 2.8 Zoll HMI Gestartet  ");
    Serial.println(" LED an GPIO 32 mit Slider h0 Dimm-Funktion      ");
    Serial.println("=================================================");

    // Initiale Werte auf dem Nextion Display setzen
    nextion_set_text("t0", "ESP32 Start");
    nextion_set_text("t1", "LED: AN (50%)");
    nextion_set_number("bt0", 1);
    nextion_set_number("h0", 50); // Slider h0 auf 50%
    nextion_set_number("n0", 0);
    nextion_set_number("j0", 0);
}

static uint32_t last_sensor_time = 0;
static float sim_temperature = 23.4;
static int sim_rpm = 1450;
static int sim_progress = 65;

void loop() {
    // 1. Laufend Touch-Ereignisse und Slider-Werte vom Nextion abfragen
    handle_nextion_data();

    // 2. Alle 1000 ms neue Sensorwerte an Nextion senden
    if (millis() - last_sensor_time >= 1000) {
        last_sensor_time = millis();

        // Messwerte simulieren
        sim_temperature += ((random(0, 20) - 10) / 10.0);
        if (sim_temperature < 18.0) sim_temperature = 18.0;
        if (sim_temperature > 35.0) sim_temperature = 35.0;

        sim_rpm += (random(-50, 50));
        sim_progress = (sim_progress + 5) % 105;

        // An Nextion senden
        char tempStr[16];
        snprintf(tempStr, sizeof(tempStr), "%.1f C", sim_temperature);
        nextion_set_text("t0", tempStr);
        nextion_set_number("n0", sim_rpm);
        nextion_set_number("j0", sim_progress);

        Serial.print("[SENSOR] Temp: ");
        Serial.print(tempStr);
        Serial.print(" | RPM: ");
        Serial.print(sim_rpm);
        Serial.print(" | Progress: ");
        Serial.print(sim_progress);
        Serial.print("% | LED: ");
        Serial.print(led_state ? "AN" : "AUS");
        Serial.print(" (");
        Serial.print(led_brightness);
        Serial.println("%)");
    }
}
